Project gojags_asrama {

  Note: 'Aplikasi Manajemen Asrama GOJAGS (Laravel 10 API + Nuxt 4 SPA). Semua waktu disimpan dalam UTC.'

// ---------- Autentikasi & Akun ----------

Table users {
  id bigint [pk, increment]
  name varchar(255) [not null]
  email varchar(255) [unique, not null]
  email_verified_at timestamp [null]
  password varchar(255) [not null]
  remember_token varchar(100) [null]
  role varchar(20) [not null, default: 'resident', note: 'enum Role: admin | resident']
  cohort varchar(100) [null, note: 'angkatan; disimpan ternormalisasi (lowercase, spasi dirapikan)']
  gender varchar(10) [null, note: 'enum Gender: male | female']
  religion varchar(50) [null, note: 'agama; disimpan ternormalisasi']
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table password_reset_tokens {
  email varchar(255) [pk]
  token varchar(255) [not null]
  created_at timestamp [null]
}

Table sessions {
  id varchar(255) [pk]
  user_id bigint [null, note: 'index saja, tanpa FK constraint']
  ip_address varchar(45) [null]
  user_agent text [null]
  payload longtext [not null]
  last_activity integer [not null, note: 'index']
}

// ---------- Antrian & Cache (bawaan Laravel) ----------

Table cache {
  key varchar(255) [pk]
  value mediumtext [not null]
  expiration integer [not null, note: 'index']
}

Table cache_locks {
  key varchar(255) [pk]
  owner varchar(255) [not null]
  expiration integer [not null, note: 'index']
}

Table jobs {
  id bigint [pk, increment]
  queue varchar(255) [not null, note: 'index']
  payload longtext [not null]
  attempts tinyint [not null]
  reserved_at integer [null]
  available_at integer [not null]
  created_at integer [not null]
}

Table job_batches {
  id varchar(255) [pk]
  name varchar(255) [not null]
  total_jobs integer [not null]
  pending_jobs integer [not null]
  failed_jobs integer [not null]
  failed_job_ids longtext [not null]
  options mediumtext [null]
  cancelled_at integer [null]
  created_at integer [not null]
  finished_at integer [null]
}

Table failed_jobs {
  id bigint [pk, increment]
  uuid varchar(255) [unique, not null]
  connection text [not null]
  queue text [not null]
  payload longtext [not null]
  exception longtext [not null]
  failed_at timestamp [not null, default: `CURRENT_TIMESTAMP`]
}

// ---------- Kamar & Hunian ----------

Table rooms {
  id bigint [pk, increment]
  number varchar(10) [unique, not null, note: '301 … 526']
  floor tinyint [not null, note: 'index; 3, 4, atau 5']
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table room_slots {
  id bigint [pk, increment]
  room_id bigint [not null, ref: > rooms.id, note: 'cascade on delete']
  position tinyint [not null, note: '1 atau 2']
  user_id bigint [null, unique, ref: > users.id, note: 'null on delete; satu user maksimal satu slot']
  checked_in_at timestamp [null]
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (room_id, position) [unique]
  }
}

Table stays {
  id bigint [pk, increment]
  user_id bigint [null, ref: > users.id, note: 'null on delete']
  room_id bigint [null, ref: > rooms.id, note: 'null on delete']
  user_name varchar(255) [not null, note: 'salinan saat check-in']
  room_number varchar(10) [not null, note: 'salinan saat check-in']
  floor tinyint [not null, note: 'salinan saat check-in']
  checked_in_at timestamp [not null]
  checked_out_at timestamp [null]
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (user_id, checked_out_at)
  }
}

// ---------- Layanan: Laundry ----------

Table laundry_orders {
  id bigint [pk, increment]
  user_id bigint [not null, ref: > users.id, note: 'cascade on delete']
  items varchar(255) [not null, note: 'deskripsi pakaian/barang']
  set_count integer [not null, note: 'jumlah stel']
  status varchar(20) [not null, default: 'pending', note: 'enum LaundryStatus: pending | done']
  cost decimal(12,2) [null, note: 'wajib saat status = done']
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (user_id, status)
  }
}

// ---------- Layanan: Catering ----------

Table catering_menus {
  id bigint [pk, increment]
  menu_number integer [not null, note: 'ditampilkan sebagai "Menu N"']
  dish_name varchar(255) [not null, note: 'nama makanan']
  meal_time varchar(20) [not null, note: 'enum MealTime: breakfast | lunch | dinner']
  provider varchar(255) [not null, note: 'penyedia']
  served_at datetime [not null, note: 'waktu catering dalam UTC']
  status varchar(20) [not null, default: 'scheduled', note: 'enum CateringStatus: scheduled | preparing | done']
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table catering_ratings {
  id bigint [pk, increment]
  catering_menu_id bigint [not null, ref: > catering_menus.id, note: 'cascade on delete']
  user_id bigint [not null, ref: > users.id, note: 'cascade on delete']
  rating tinyint [not null, note: '1..5']
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (catering_menu_id, user_id) [unique]
  }
}

// ---------- Layanan: Izin Keluar ----------

Table leave_permits {
  id bigint [pk, increment]
  user_id bigint [not null, ref: > users.id, note: 'cascade on delete']
  leaves_at datetime [not null]
  returns_at datetime [not null, note: 'returns_at > leaves_at']
  reason text [not null]
  status varchar(20) [not null, default: 'pending', note: 'enum PermitStatus: pending | approved | rejected']
  reviewed_by bigint [null, ref: > users.id, note: 'admin; null on delete']
  reviewed_at timestamp [null]
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (user_id, status)
  }
}

// ---------- Layanan: Keluhan Asrama ----------

Table complaints {
  id bigint [pk, increment]
  user_id bigint [not null, ref: > users.id, note: 'cascade on delete']
  title varchar(150) [not null]
  body text [not null]
  status varchar(20) [not null, default: 'pending', note: 'enum ComplaintStatus: pending | resolved']
  resolved_by bigint [null, ref: > users.id, note: 'admin; null on delete']
  resolved_at timestamp [null]
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (user_id, status)
  }
}
