Project gojags_asrama {
  database_type: 'SQLite'
  Note: 'Aplikasi Asrama (Laravel 10). Tabel inti: users, kamar, kamar_slots, kamar_stays, laundry, catering, catering_ratings, izin_keluar, keluhan_asrama.'
}

// ---------- Autentikasi & Akun ----------

Table users {
  id bigint [pk, increment]
  name varchar
  email varchar [unique, not null]
  email_verified_at timestamp [null]
  password varchar
  remember_token varchar [null]
  role varchar [not null, default: 'user', note: 'user | admin']
  angkatan varchar(100) [null, note: 'Profil untuk alokasi kamar']
  gender varchar(20) [null, note: 'Laki-laki | Perempuan']
  agama varchar(50) [null]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table password_reset_tokens {
  email varchar [pk]
  token varchar
  created_at timestamp [null]
}

Table sessions {
  id varchar [pk]
  user_id bigint [null, note: 'index saja, tanpa FK constraint']
  ip_address varchar(45) [null]
  user_agent text [null]
  payload text
  last_activity integer [note: 'index']
}

// ---------- Antrian & Cache (bawaan Laravel) ----------

Table cache {
  key varchar [pk]
  value text
  expiration integer [note: 'index']
}

Table cache_locks {
  key varchar [pk]
  owner varchar
  expiration integer [note: 'index']
}

Table jobs {
  id bigint [pk, increment]
  queue varchar [note: 'index']
  payload text
  attempts tinyint
  reserved_at integer [null]
  available_at integer
  created_at integer
}

Table job_batches {
  id varchar [pk]
  name varchar
  total_jobs integer
  pending_jobs integer
  failed_jobs integer
  failed_job_ids text
  options text [null]
  cancelled_at integer [null]
  created_at integer
  finished_at integer [null]
}

Table failed_jobs {
  id bigint [pk, increment]
  uuid varchar [unique]
  connection text
  queue text
  payload text
  exception text
  failed_at timestamp [default: `CURRENT_TIMESTAMP`]
}

// ---------- Kamar ----------

Table kamar {
  id bigint [pk, increment]
  name varchar [note: 'Nomor kamar, mis. 501']
  status varchar [note: 'Kosong | Assigned | Terisi | Penuh']
  lantai tinyint [default: 3, note: '3, 4, atau 5']
  date date
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table kamar_slots {
  id bigint [pk, increment]
  kamar_id bigint [not null, ref: > kamar.id, note: 'cascade on delete']
  nomor tinyint [note: 'Slot 1 atau 2 (max 2 penghuni/kamar)']
  user_id bigint [null, unique, ref: > users.id, note: 'null on delete']
  checked_in_at timestamp [null]
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (kamar_id, nomor) [unique]
  }
}

Table kamar_stays {
  id bigint [pk, increment]
  user_id bigint [null, ref: > users.id, note: 'null on delete']
  kamar_id bigint [null, ref: > kamar.id, note: 'null on delete']
  user_name varchar [note: 'snapshot nama user']
  kamar_name varchar [note: 'snapshot nomor kamar']
  lantai tinyint
  checked_in_at timestamp
  checked_out_at timestamp [null]
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (user_id, checked_out_at) [name: 'kamar_stays_user_id_checked_out_at_index']
  }
}

// ---------- Layanan: Laundry ----------

Table laundry {
  id bigint [pk, increment]
  name varchar [note: 'snapshot nama pemesan']
  detail varchar [note: 'deskripsi pakaian']
  status varchar [note: 'Diajukan | Selesai']
  date date
  user_id bigint [null, ref: > users.id, note: 'null on delete']
  jumlah_stel integer [null]
  biaya decimal(12,2) [null, note: 'wajib saat status Selesai']
  created_at timestamp [null]
  updated_at timestamp [null]
}

// ---------- Layanan: Catering ----------

Table catering {
  id bigint [pk, increment]
  name varchar [note: 'mis. "Menu 3"']
  detail varchar [note: 'snapshot penyedia']
  status varchar [note: 'Terjadwal | Disiapkan | Selesai']
  date date
  menu integer [null]
  penyedia varchar(255) [null]
  waktu_catering datetime [null, note: 'disimpan dalam UTC']
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table catering_ratings {
  id bigint [pk, increment]
  catering_id bigint [not null, ref: > catering.id, note: 'cascade on delete']
  user_id bigint [not null, ref: > users.id, note: 'cascade on delete']
  rating tinyint [note: '1..5']
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (catering_id, user_id) [unique]
  }
}

// ---------- Layanan: Izin Keluar ----------

Table izin_keluar {
  id bigint [pk, increment]
  user_id bigint [not null, ref: > users.id, note: 'cascade on delete']
  waktu_keluar datetime
  waktu_kembali datetime [note: 'harus setelah waktu_keluar']
  alasan text
  status varchar(20) [default: 'Diajukan', note: 'Diajukan | Disetujui | Ditolak']
  reviewed_by bigint [null, ref: > users.id, note: 'admin; null on delete']
  reviewed_at timestamp [null]
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (user_id, status) [name: 'izin_keluar_user_id_status_index']
  }
}

// ---------- Layanan: Keluhan Asrama ----------

Table keluhan_asrama {
  id bigint [pk, increment]
  user_id bigint [not null, ref: > users.id, note: 'cascade on delete']
  judul varchar(150)
  isi text
  status varchar(20) [default: 'Diajukan', note: 'Diajukan | Resolved']
  resolved_by bigint [null, ref: > users.id, note: 'admin; null on delete']
  resolved_at timestamp [null]
  created_at timestamp [null]
  updated_at timestamp [null]

  indexes {
    (user_id, status) [name: 'keluhan_asrama_user_id_status_index']
  }
}
